f_detect_market() =>
    string p = syminfo.prefix
    string t = syminfo.type
    string tk = syminfo.ticker
    string cur = syminfo.currency

    bool prefixA = p == "SSE" or p == "SZSE" or p == "SH" or p == "SZ" or p == "BSE" or p == "BJ"
    bool currencyA = cur == "CNY" and (t == "stock" or t == "fund" or t == "index")
    bool suffixA = str.endswith(tk, ".SS") or str.endswith(tk, ".SZ") or str.endswith(tk, ".SH") or str.endswith(tk, ".BJ")
    bool detectedA = prefixA or currencyA or suffixA
    bool detectedCrypto = p == "BINANCE" or p == "OKX" or p == "BYBIT" or p == "COINBASE" or p == "KUCOIN" or p == "BITGET" or p == "GATEIO" or p == "MEXC" or t == "crypto"
    bool detectedUS = p == "NASDAQ" or p == "NYSE" or p == "AMEX" or p == "BATS" or p == "ARCA"

    string result = "Other Equities"

    if detectedA
        result := "China A-Shares"
    else if detectedCrypto
        result := "Crypto"
    else if detectedUS
        result := "US Equities"

    result

string marketType = marketModeInput

    marketType := f_detect_market()

var int latestFractalType = FRACTAL_NONE

var float latestFractalPrice = na

var int latestFractalBar = na

bool newStrokeCreated = false

if rawTopFractal and barstate.isconfirmed
    int candidateBar = bar_index - 1
    float candidatePrice = high[1]
    int n = array.size(strokePrices)

    if n == 0
        array.push(strokePrices, candidatePrice)
        array.push(strokeTypes, FRACTAL_TOP)
        array.push(strokeBars, candidateBar)

        latestFractalType := FRACTAL_TOP
        latestFractalPrice := candidatePrice
        latestFractalBar := candidateBar

        newStrokeCreated := true

    else
        int lastType = array.get(strokeTypes, n - 1)
        float lastPrice = array.get(strokePrices, n - 1)
        int lastBar = array.get(strokeBars, n - 1)

        if lastType == FRACTAL_TOP
            if candidatePrice > lastPrice
                array.set(strokePrices, n - 1, candidatePrice)
                array.set(strokeBars, n - 1, candidateBar)

                latestFractalType := FRACTAL_TOP
                latestFractalPrice := candidatePrice
                latestFractalBar := candidateBar

        else
            bool enoughBars = candidateBar - lastBar >= pStrokeMinBars
            bool enoughMove = not na(fractalAtr) and candidatePrice - lastPrice >= fractalAtr * pStrokeAtr

            if enoughBars and enoughMove
                array.push(strokePrices, candidatePrice)
                array.push(strokeTypes, FRACTAL_TOP)
                array.push(strokeBars, candidateBar)

                latestFractalType := FRACTAL_TOP
                latestFractalPrice := candidatePrice
                latestFractalBar := candidateBar

                newStrokeCreated := true

if rawBottomFractal and barstate.isconfirmed
    int candidateBar = bar_index - 1
    float candidatePrice = low[1]
    int n = array.size(strokePrices)

    if n == 0
        array.push(strokePrices, candidatePrice)
        array.push(strokeTypes, FRACTAL_BOTTOM)
        array.push(strokeBars, candidateBar)

        latestFractalType := FRACTAL_BOTTOM
        latestFractalPrice := candidatePrice
        latestFractalBar := candidateBar

        newStrokeCreated := true

    else
        int lastType = array.get(strokeTypes, n - 1)
        float lastPrice = array.get(strokePrices, n - 1)
        int lastBar = array.get(strokeBars, n - 1)

        if lastType == FRACTAL_BOTTOM
            if candidatePrice < lastPrice
                array.set(strokePrices, n - 1, candidatePrice)
                array.set(strokeBars, n - 1, candidateBar)

                latestFractalType := FRACTAL_BOTTOM
                latestFractalPrice := candidatePrice
                latestFractalBar := candidateBar

        else
            bool enoughBars = candidateBar - lastBar >= pStrokeMinBars
            bool enoughMove = not na(fractalAtr) and lastPrice - candidatePrice >= fractalAtr * pStrokeAtr

            if enoughBars and enoughMove
                array.push(strokePrices, candidatePrice)
                array.push(strokeTypes, FRACTAL_BOTTOM)
                array.push(strokeBars, candidateBar)

                latestFractalType := FRACTAL_BOTTOM
                latestFractalPrice := candidatePrice
                latestFractalBar := candidateBar

while array.size(strokePrices) > maxStoredStrokes

            prevZsZD := zsZD

            zsZD := currentZD
